/**
 * @param {Array<unknown>} iterable
 * @param {(value: unknown) => Promise<unknown>} callbackFn
 * @param {number} [size=Infinity]
 *
 * @return {Promise<Array<unknown>>}
 */
export default async function mapAsyncLimit<T, U>(
  iterable: Array<T>,
  callbackFn: (value: T) => Promise<U>,
  size?: number,
): Promise<Array<U>> {
  if (size === undefined) {
    return Promise.all(iterable.map(callbackFn));
  }

  const result: Array<U> = [];

  //按照size 大小，来分块
  for (let i = 0; i < iterable.length; i += size) {
    //截取当前批次的元素
    const chunk = iterable.slice(i, i + size);

    //将当前chunk 的元素隐射为promise的数组
    const chunkPromises = chunk.map(callbackFn);

    //使用promise.all 等待当前chunk 全部完成
    const chunkResults = await Promise.all(chunkPromises);

    //将结果存入最后的数组
    result.push(...chunkResults);
  }

  return result;
}

 * @param {Promise<T>} promise
 * @param {number} duration
 * @return {Promise<T>}
 */
export default function promiseTimeout(promise, duration) {
  //create a timeout promise, that must be reject.
  const promiseTimer = new Promise((_, reject) => {
    setTimeout(() => reject("Promise timeout"), duration);
  });
  //let them have the race
  return Promise.race([promise, promiseTimer]);
}

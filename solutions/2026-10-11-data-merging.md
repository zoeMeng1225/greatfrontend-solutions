/**
 * @param {Array<{user: number, duration: number, equipment: Array<string>}>} sessions
 * @return {Array}
 */
export default function mergeData(sessions) {
  const map = new Map<number, Session>();

  for (const session of sessions) {
    //check if this user is existing?
    if (map.has(session.user)) {
      //if existing： combine
      const existing = map.get(session.user)!;
      existing.duration += session.duration;

      //equipment
      const combineEquipment = [...session.equipment, ...existing.equipment];
      const makeNewEquipments = Array.from(new Set(combineEquipment)).sort();
      existing.equipment = makeNewEquipments;
    } else {
      map.set(session.user, {
        user: session.user,
        duration: session.duration,
        //just in case
        equipment: [...session.equipment].sort(),
      });
    }
  }
  console.log(map.values());
  return Array.from(map.values());
}

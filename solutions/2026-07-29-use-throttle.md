 * @template T
 * @param {T} value
 * @param {number} interval

import { useState, useEffect, useRef } from "react";

export default function useThrottle(value, interval = 500) {
  //create a state to store the throttled value
  const [throttledValue, setThrottledValue] = useState(value);

  //use useRef to track the last time the value was updated
  const lastUpdated = useRef(Date.now());

  useEffect(() => {
    const elapsed = Date.now() - lastUpdated.current;

    //if enough time has passed, udpate immediately
    if (elapsed >= interval) {
      setThrottledValue(value);
      lastUpdated.current = Date.now();
    } else {
      //if traggered too soon, set a timer for the remaining time
      const timer = setTimeout(() => {
        setThrottledValue(value);
        lastUpdated.current = Date.now();
      }, interval - elapsed);

      //cleanup: clear the timer if the value change again
      return () => {
        clearTimeout(timer);
      };
    }
  }, [value, interval]);

  return throttledValue;
}

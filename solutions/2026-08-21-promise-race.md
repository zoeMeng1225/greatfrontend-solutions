/**
 * @param {Array} iterable
 * @return {Promise}
 */
export default function promiseRace(iterable) {
  return new Promise((resolve, reject) => {
    //if iteranle is empty, the promise returned will be forever pending
    if (iterable.length === 0) return; //所以这里直接 return，不调用 resolve 也不调用 reject

    //iterate all items
    iterable.forEach((item) => {
      //因为不能保证item 一定是Promise，可能是（普通数字42）
      //所以用Promise.solve(item)包一下，把它们统一变成Promise
      Promise.resolve(item).then(resolve, reject);
      //核心逻辑，谁先完成，谁就调用外面的resolve或reject
      //因为Promise的状态一旦改变就不可逆
      //所以只有第一介调用的才会生效
    });
  });
}

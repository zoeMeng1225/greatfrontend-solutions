/**
 * @param {Promise} p1
 * @param {Promise} p2
 * @return {Promise<any>}
 */
export default function promiseMerge(p1, p2) {
  //用Primise.all是因为This returned promise fulfills when both input promises fulfill"
  //（当两个输入 Promise 都 成功时，返回的 Promise 才成功。
  return Promise.all([p1, p2]).then(([val1, val2]) => {
    //number, add together
    if (typeof val1 === "number" && typeof val2 === "number") {
      return val1 + val2;
    }
    //string， concatenated
    if (typeof val1 === "string" && typeof val2 === "string") {
      return val1 + val2;
    }
    //array , combimbed
    if (Array.isArray(val1) && Array.isArray(val2)) {
      return [...val1, ...val2];
    }
    //object(plain Object), merge
    if (isPlantObject(val1) && isPlantObject(val2)) {
      return { ...val1, ...val2 };
    }
    throw "Unsupported data types";
  });
}
function isPlantObject(val) {
  //最可靠检测类型的方法，因为typeof不能区分 array、object、null 等。
  //可以检查plain object
  return Object.prototype.toString.call(val) === "[object Object]";
}

/**
 * @param {...Function} args
 * @returns Function
 */
export default function compose(...fns) {
  if (fns.length === 0) {
    return (x) => x;
  }

  return function (x) {
    return fns.reduceRight((acc, fn) => {
      return fn(acc);
    }, x);
  };
}

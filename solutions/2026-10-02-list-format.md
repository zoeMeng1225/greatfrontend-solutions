export default function listFormat(
  items: Array<string>,
  options?: { sorted?: boolean; length?: number; unique?: boolean },
): string {
  let processedItems = [...items];
  processedItems = processedItems.filter(
    (item) => item && item.trim().length > 0,
  );

  //unique
  if (options?.unique) {
    processedItems = Array.from(new Set(processedItems));
  }

  //sort
  if (options?.sorted) {
    processedItems.sort();
  }

  //empty list
  if (processedItems.length === 0) {
    return "";
  }

  //length
  if (
    options?.length &&
    options.length > 0 &&
    options?.length < processedItems.length
  ) {
    const limit = options.length;
    const visibleItems = processedItems.slice(0, limit);
    const remainCount = processedItems.length - limit;

    return `${visibleItems.join(", ")} and ${remainCount} ${remainCount <= 1 ? 'other' : 'others'}`;
  }

  if (processedItems.length === 1) {
    return processedItems[0];
  }

  const lastItem = processedItems.pop();
  return `${processedItems.join(", ")} and ${lastItem}`;
}
